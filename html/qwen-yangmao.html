#!/usr/bin/env python3
"""复现 Qwen3.8-27B 连通性测试；仅使用 Python 标准库。"""

import getpass
import json
import os
import sys
import time
import urllib.error
import urllib.request


BASE_URL = os.environ.get(
    "DASHSCOPE_BASE_URL",
    "https://ws-rp9lenauh953vhh5.cn-beijing.maas.aliyuncs.com/compatible-mode/v1",
).rstrip("/")
MODEL = os.environ.get("QWEN_MODEL", "qwen3.8-27b")


def request(path, api_key, body=None):
    """返回 HTTP 状态、端到端耗时及 JSON 响应，不输出鉴权头。"""
    data = None if body is None else json.dumps(body).encode("utf-8")
    req = urllib.request.Request(
        BASE_URL + path,
        data=data,
        headers={
            "Authorization": "Bearer " + api_key,
            "Content-Type": "application/json",
        },
    )
    start = time.monotonic()
    try:
        with urllib.request.urlopen(req, timeout=45) as response:
            status = response.status
            raw = response.read().decode("utf-8")
    except urllib.error.HTTPError as error:
        status = error.code
        raw = error.read().decode("utf-8", errors="replace")
    except (urllib.error.URLError, TimeoutError, OSError) as error:
        return {
            "status": None,
            "seconds": round(time.monotonic() - start, 2),
            "error": str(error).replace(api_key, "[REDACTED]"),
        }

    raw = raw.replace(api_key, "[REDACTED]")
    try:
        payload = json.loads(raw)
    except ValueError:
        payload = raw[:4000]
    return {
        "status": status,
        "seconds": round(time.monotonic() - start, 2),
        "body": payload,
    }


def show(label, value):
    print(label, flush=True)
    print(json.dumps(value, ensure_ascii=False, indent=2), flush=True)


def main():
    api_key = os.environ.get("DASHSCOPE_API_KEY", "").strip()
    if not api_key:
        if not sys.stdin.isatty():
            print("请设置环境变量 DASHSCOPE_API_KEY。", file=sys.stderr)
            return 2
        api_key = getpass.getpass("请输入 API Key（输入不回显）: ").strip()
    if not api_key:
        print("API Key 不能为空。", file=sys.stderr)
        return 2

    models = request("/models", api_key)
    model_body = models.get("body")
    ids = []
    if models.get("status") == 200 and isinstance(model_body, dict):
        ids = [item.get("id", "") for item in model_body.get("data", [])]
    candidates = [name for name in ids if "qwen3.8" in name.lower() and "27b" in name.lower()]
    selected_model = MODEL
    if MODEL not in ids and MODEL == "qwen3.8-27b" and candidates:
        selected_model = candidates[0]

    summary = {key: value for key, value in models.items() if key != "body"}
    summary.update({"matching_models": candidates, "selected_model": selected_model})
    if models.get("status") != 200:
        summary["body"] = model_body
    show("1. GET /models", summary)

    # /models 可能不受支持，仍直接尝试目标模型。
    payload = {
        "model": selected_model,
        "messages": [{"role": "user", "content": "只回复：连接成功"}],
        "max_tokens": 128,
        "temperature": 0,
        "enable_thinking": False,
    }
    result = request("/chat/completions", api_key, payload)
    show("2. POST /chat/completions", result)

    body = result.get("body")
    if result.get("status") != 200 or not isinstance(body, dict):
        return 1
    try:
        reply = body["choices"][0]["message"]["content"]
        returned_model = body["model"]
    except (KeyError, IndexError, TypeError):
        print("响应缺少 model 或 choices[0].message.content。", file=sys.stderr)
        return 1
    if not isinstance(reply, str) or not reply.strip():
        print("响应正文为空。", file=sys.stderr)
        return 1
    show("3. 调用成功", {"model": returned_model, "reply": reply, "usage": body.get("usage")})
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
